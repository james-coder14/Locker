rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Must be signed in with a verified Google account.
    // TO LOCK TO YOUR SCHOOL: add  && request.auth.token.email.matches('.*@yourschool[.]org')
    // (Note: this only works if your school's address is the one used to sign up.)
    function signedIn() {
      return request.auth != null && request.auth.token.email_verified == true;
    }

    // Moderators are users who have a document at mods/<their uid>.
    // Only you can create these, in the Firebase console.
    function isMod() {
      return signedIn() && exists(/databases/$(database)/documents/mods/$(request.auth.uid));
    }

    function validChannel(c) {
      return c in ['Student Body', 'Juniors', 'Student Council', 'Math 1A', 'Girls Soccer'];
    }

    match /mods/{uid} {
      allow get: if signedIn() && request.auth.uid == uid;
      allow list, write: if false;
    }

    match /posts/{id} {
      allow read: if signedIn();
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['text', 'author', 'authorName', 'channel', 'createdAt'])
        && request.resource.data.author == request.auth.uid
        && request.resource.data.authorName is string
        && request.resource.data.authorName.size() > 0
        && request.resource.data.authorName.size() <= 40
        && request.resource.data.text is string
        && request.resource.data.text.size() > 0
        && request.resource.data.text.size() <= 500
        && validChannel(request.resource.data.channel)
        && request.resource.data.createdAt == request.time;
      allow update: if false;
      allow delete: if signedIn() && (resource.data.author == request.auth.uid || isMod());
    }

    match /likes/{id} {
      allow read: if signedIn();
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['post', 'user', 'channel'])
        && request.resource.data.user == request.auth.uid
        && id == request.resource.data.post + '_' + request.auth.uid
        && validChannel(request.resource.data.channel);
      allow update: if false;
      allow delete: if signedIn() && resource.data.user == request.auth.uid;
    }

    match /reports/{id} {
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['post', 'by', 'text', 'channel', 'at'])
        && request.resource.data.by == request.auth.uid
        && request.resource.data.text is string
        && request.resource.data.text.size() <= 500;
      allow read, delete: if isMod();
      allow update: if false;
    }

    match /events/{id} {
      allow read: if signedIn();
      allow write: if isMod();
    }
  }
}
