rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null && request.auth.token.email_verified == true;
    }
    function isMod() {
      return signedIn() && exists(/databases/$(database)/documents/mods/$(request.auth.uid));
    }
    function isTeacher() {
      return signedIn() && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'teacher';
    }
    function isStudent() {
      return signedIn() && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'student';
    }
    function isMember(classId) {
      return signedIn() && exists(/databases/$(database)/documents/members/$(classId + '_' + request.auth.uid));
    }
    function isOwner(classId) {
      return signedIn() && get(/databases/$(database)/documents/classes/$(classId)).data.ownerUid == request.auth.uid;
    }
    function isAdmin(classId) {
      return signedIn() && exists(/databases/$(database)/documents/members/$(classId + '_' + request.auth.uid))
        && get(/databases/$(database)/documents/members/$(classId + '_' + request.auth.uid)).data.role == 'admin';
    }
    function canModerate(classId) {
      return isOwner(classId) || isAdmin(classId);
    }
    function otherOf(p) {
      return p[0] == request.auth.uid ? p[1] : p[0];
    }
    function convParts(convId) {
      return get(/databases/$(database)/documents/conversations/$(convId)).data.participants;
    }
    function matesOK(a, b) {
      return (exists(/databases/$(database)/documents/mates/$(a + '_' + b))
          && get(/databases/$(database)/documents/mates/$(a + '_' + b)).data.status == 'accepted')
        || (exists(/databases/$(database)/documents/mates/$(b + '_' + a))
          && get(/databases/$(database)/documents/mates/$(b + '_' + a)).data.status == 'accepted');
    }
    // true if either person has blocked the other
    function blockedEither(a, b) {
      return exists(/databases/$(database)/documents/blocks/$(a + '_' + b))
        || exists(/databases/$(database)/documents/blocks/$(b + '_' + a));
    }

    match /mods/{uid} {
      allow get: if signedIn() && request.auth.uid == uid;
      allow list, write: if false;
    }

    match /usernames/{name} {
      allow get: if signedIn();
      allow create: if signedIn() && name.matches('^[a-z0-9_]{3,20}$')
        && request.resource.data.keys().hasOnly(['uid'])
        && request.resource.data.uid == request.auth.uid;
      allow update, delete: if signedIn() && (resource.data.uid == request.auth.uid || isMod());
    }

    match /users/{uid} {
      allow read: if signedIn();
      // Full profile create/update
      allow create, update: if signedIn() && request.auth.uid == uid
        && request.resource.data.name is string && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 40
        && request.resource.data.bio is string && request.resource.data.bio.size() <= 300
        && request.resource.data.contactEmail is string && request.resource.data.contactEmail.size() <= 100
        && request.resource.data.username is string && request.resource.data.username.matches('^[a-z0-9_]{3,20}$')
        && request.resource.data.role in ['teacher', 'student']
        // ROLE LOCK: cannot change role once onboarded
        && (!resource.exists || !resource.data.get('onboarded', false) || request.resource.data.role == resource.data.role)
        && request.resource.data.get('theme', 'default') in ['default', 'ocean', 'forest', 'sunset', 'rose']
        && request.resource.data.photo is string && request.resource.data.photo.size() <= 150000
        && request.resource.data.onboarded is bool
        && request.resource.data.get('status', '') is string
        && request.resource.data.get('status', '').size() <= 40;
      // Heartbeat only (lastSeen)
      allow update: if signedIn() && request.auth.uid == uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['lastSeen']);
      // Mod: shadowban flag only
      allow update: if isMod()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['shadowbanned']);
      allow delete: if isMod();
    }

    // Blocks: only the blocker can see and manage their own blocks
    match /blocks/{id} {
      allow read: if signedIn() && resource.data.blocker == request.auth.uid;
      allow create: if signedIn() && request.resource.data.blocker == request.auth.uid
        && request.resource.data.keys().hasOnly(['blocker', 'blocked', 'at'])
        && request.resource.data.blocked != request.auth.uid
        && id == request.resource.data.blocker + '_' + request.resource.data.blocked;
      allow update: if false;
      allow delete: if signedIn() && resource.data.blocker == request.auth.uid;
    }

    match /classes/{classId} {
      allow read: if isMember(classId) || isMod();
      allow create: if (isTeacher() || isMod()) && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.keys().hasOnly(['name', 'code', 'ownerUid', 'ownerName', 'createdAt'])
        && request.resource.data.name is string && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 60
        && request.resource.data.code is string && request.resource.data.code.size() == 6;
      allow update, delete: if signedIn() && (resource.data.ownerUid == request.auth.uid || isMod());
    }

    match /classCodes/{code} {
      allow get: if signedIn();
      allow create: if signedIn() && request.resource.data.keys().hasOnly(['classId'])
        && isOwner(request.resource.data.classId);
      allow delete: if signedIn() && (isOwner(resource.data.classId) || isMod());
    }

    match /members/{id} {
      allow read: if signedIn() && (resource.data.uid == request.auth.uid || isMember(resource.data.classId) || isMod());
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['classId', 'uid', 'role', 'joinedAt'])
        && id == request.resource.data.classId + '_' + request.auth.uid
        && request.resource.data.uid == request.auth.uid
        && (request.resource.data.role == 'student' || (request.resource.data.role == 'owner' && isOwner(request.resource.data.classId)));
      allow update: if isOwner(resource.data.classId) && resource.data.role != 'owner'
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['role'])
        && request.resource.data.role in ['student', 'admin'];
      allow delete: if signedIn() && ((resource.data.uid == request.auth.uid && resource.data.role == 'student') || isOwner(resource.data.classId) || isMod());
    }

    match /mates/{id} {
      allow read: if signedIn() && (request.auth.uid in resource.data.participants || isMod());
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['participants', 'names', 'status', 'requestedBy', 'createdAt'])
        && request.resource.data.participants.size() == 2
        && request.resource.data.participants[0] == request.auth.uid
        && request.resource.data.participants[1] != request.auth.uid
        && request.resource.data.requestedBy == request.auth.uid
        && request.resource.data.status == 'pending'
        && id == request.resource.data.participants[0] + '_' + request.resource.data.participants[1]
        && !blockedEither(request.resource.data.participants[0], request.resource.data.participants[1])
        && exists(/databases/$(database)/documents/users/$(request.resource.data.participants[1]));
      allow update: if signedIn() && request.auth.uid in resource.data.participants
        && resource.data.status == 'pending'
        && resource.data.requestedBy != request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status'])
        && request.resource.data.status == 'accepted';
      allow delete: if signedIn() && request.auth.uid in resource.data.participants;
    }

    match /posts/{id} {
      allow read: if isMember(resource.data.classId);
      allow create: if isMember(request.resource.data.classId)
        && request.resource.data.keys().hasOnly(['text', 'author', 'authorName', 'classId', 'createdAt'])
        && request.resource.data.author == request.auth.uid
        && request.resource.data.authorName is string && request.resource.data.authorName.size() > 0 && request.resource.data.authorName.size() <= 40
        && request.resource.data.text is string && request.resource.data.text.size() > 0 && request.resource.data.text.size() <= 2000
        && request.resource.data.createdAt == request.time;
      allow update: if isMod()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['pinned']);
      allow delete: if signedIn() && (resource.data.author == request.auth.uid || canModerate(resource.data.classId) || isMod());
    }

    match /likes/{id} {
      allow read: if isMember(resource.data.classId);
      allow create: if isMember(request.resource.data.classId)
        && request.resource.data.keys().hasOnly(['post', 'user', 'classId'])
        && request.resource.data.user == request.auth.uid
        && id == request.resource.data.post + '_' + request.auth.uid;
      allow update: if false;
      allow delete: if signedIn() && resource.data.user == request.auth.uid;
    }

    match /reports/{id} {
      allow create: if isMember(request.resource.data.classId)
        && request.resource.data.keys().hasOnly(['post', 'by', 'text', 'classId', 'at'])
        && request.resource.data.by == request.auth.uid
        && request.resource.data.text is string && request.resource.data.text.size() <= 500;
      allow read, delete: if canModerate(resource.data.classId) || isMod();
      allow update: if false;
    }

    match /dmReports/{id} {
      allow create: if signedIn() && request.resource.data.by == request.auth.uid
        && request.resource.data.keys().hasOnly(['by', 'convId', 'messageId', 'text', 'reason', 'at'])
        && request.resource.data.text is string && request.resource.data.text.size() <= 500
        && request.resource.data.reason is string && request.resource.data.reason.size() <= 300
        && request.auth.uid in get(/databases/$(database)/documents/conversations/$(request.resource.data.convId)).data.participants;
      allow read, delete: if isMod();
      allow update: if false;
    }

    match /events/{id} {
      allow read: if isMember(resource.data.classId);
      allow create: if canModerate(request.resource.data.classId)
        && request.resource.data.keys().hasOnly(['classId', 'title', 'when', 'at'])
        && request.resource.data.title is string && request.resource.data.title.size() > 0 && request.resource.data.title.size() <= 100;
      allow update: if false;
      allow delete: if canModerate(resource.data.classId);
    }

    match /conversations/{convId} {
      allow read: if signedIn() && (request.auth.uid in resource.data.participants || isMod());
      // 1:1 or group (mates) conversations
      allow create: if signedIn()
        && (isStudent() || isTeacher())
        && request.auth.uid in request.resource.data.participants
        && request.resource.data.participants.size() >= 2
        && request.resource.data.participants.size() <= 12
        && request.resource.data.lastText is string && request.resource.data.lastText.size() <= 80
        && request.resource.data.typing == {}
        && (
          // Classic 1:1 chat (any student/teacher to any student/teacher, if mates or same class)
          (request.resource.data.participants.size() == 2
            && !request.resource.data.keys().hasAny(['isGroup'])
            && request.resource.data.keys().hasOnly(['participants', 'classId', 'names', 'createdAt', 'lastAt', 'lastText', 'typing'])
            && !blockedEither(request.resource.data.participants[0], request.resource.data.participants[1])
            && get(/databases/$(database)/documents/users/$(otherOf(request.resource.data.participants))).data.role in ['student', 'teacher']
            && (
              (exists(/databases/$(database)/documents/members/$(request.resource.data.classId + '_' + request.auth.uid))
                && exists(/databases/$(database)/documents/members/$(request.resource.data.classId + '_' + otherOf(request.resource.data.participants))))
              || matesOK(request.resource.data.participants[0], request.resource.data.participants[1])
            )
          )
          ||
          // Group chat
          (request.resource.data.get('isGroup', false) == true
            && request.resource.data.createdBy == request.auth.uid
            && request.resource.data.keys().hasOnly(['participants', 'classId', 'names', 'createdAt', 'lastAt', 'lastText', 'typing', 'isGroup', 'title', 'createdBy', 'color'])
            && request.resource.data.title is string
            && request.resource.data.title.size() > 0
            && request.resource.data.title.size() <= 40
          )
        );
      allow update: if signedIn() && request.auth.uid in resource.data.participants
        && (
          // normal chat activity
          (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['lastAt', 'lastText', 'typing'])
            && (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['typing'])
              || request.resource.data.typing.diff(resource.data.typing).affectedKeys().hasOnly([request.auth.uid]))
            && request.resource.data.lastText.size() <= 80)
          ||
          // leave group / remove member (participants + names only)
          (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['participants', 'names'])
            && request.auth.uid in resource.data.participants
            && request.resource.data.participants is list
            && request.resource.data.participants.size() >= 1
            && request.resource.data.participants.size() <= resource.data.participants.size())
        );
      allow delete: if signedIn() && (request.auth.uid in resource.data.participants || isMod());

      match /messages/{msgId} {
        allow read: if signedIn() && (request.auth.uid in convParts(convId) || isMod());
        allow create: if signedIn() && request.auth.uid in convParts(convId)
          && request.resource.data.from == request.auth.uid
          && request.resource.data.keys().hasOnly(['from', 'type', 'text', 'gifUrl', 'createdAt', 'editedAt', 'readBy', 'deliveredTo', 'reactions'])
          && request.resource.data.type in ['text', 'gif']
          && request.resource.data.text is string && request.resource.data.text.size() <= 500
          && (request.resource.data.gifUrl == null || (request.resource.data.gifUrl is string && request.resource.data.gifUrl.matches('^https://.*')))
          && request.resource.data.createdAt == request.time
          && request.resource.data.editedAt == null
          && request.resource.data.readBy == {}
          && request.resource.data.deliveredTo == {}
          && request.resource.data.reactions == {};
        allow update: if signedIn() && request.auth.uid in convParts(convId) && (
          (resource.data.from == request.auth.uid
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['text', 'editedAt'])
            && request.resource.data.text.size() <= 500)
          ||
          (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['readBy', 'deliveredTo', 'reactions'])
            && request.resource.data.readBy.diff(resource.data.readBy).affectedKeys().hasOnly([request.auth.uid])
            && request.resource.data.deliveredTo.diff(resource.data.deliveredTo).affectedKeys().hasOnly([request.auth.uid])
            && request.resource.data.reactions.diff(resource.data.reactions).affectedKeys().hasOnly([request.auth.uid])));
        allow delete: if isMod();
      }
    }

    match /comments/{id} {
      allow read: if signedIn() && isMember(resource.data.classId);
      allow create: if signedIn() && isMember(request.resource.data.classId)
        && request.resource.data.keys().hasOnly(['postId', 'classId', 'author', 'authorName', 'text', 'createdAt', 'parentId'])
        && request.resource.data.author == request.auth.uid
        && request.resource.data.text is string && request.resource.data.text.size() > 0 && request.resource.data.text.size() <= 400
        && request.resource.data.createdAt == request.time
        && (request.resource.data.parentId == null || request.resource.data.parentId is string);
      allow delete: if signedIn() && (resource.data.author == request.auth.uid || canModerate(resource.data.classId) || isMod());
      allow update: if false;
    }

    match /polls/{id} {
      allow read: if signedIn() && isMember(resource.data.classId);
      allow create: if signedIn() && isMember(request.resource.data.classId)
        && request.resource.data.keys().hasOnly(['classId', 'author', 'authorName', 'question', 'options', 'votes', 'createdAt'])
        && request.resource.data.author == request.auth.uid
        && request.resource.data.question is string && request.resource.data.question.size() > 0 && request.resource.data.question.size() <= 120
        && request.resource.data.options is list && request.resource.data.options.size() >= 2 && request.resource.data.options.size() <= 4
        && request.resource.data.votes == {}
        && request.resource.data.createdAt == request.time;
      allow update: if signedIn() && isMember(resource.data.classId)
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['votes'])
        && request.resource.data.votes.diff(resource.data.votes).affectedKeys().hasOnly([request.auth.uid]);
      allow delete: if signedIn() && (resource.data.author == request.auth.uid || canModerate(resource.data.classId) || isMod());
    }

    match /mutes/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || isMod());
      allow create, update: if isMod()
        && request.resource.data.keys().hasOnly(['until', 'reason', 'by', 'at'])
        && request.resource.data.by == request.auth.uid
        && request.resource.data.until is int
        && request.resource.data.reason is string && request.resource.data.reason.size() <= 200;
      allow delete: if isMod();
    }

    match /warnings/{id} {
      allow read: if isMod();
      allow create: if isMod()
        && request.resource.data.keys().hasOnly(['uid', 'text', 'by', 'byLabel', 'at'])
        && request.resource.data.by == request.auth.uid
        && request.resource.data.uid is string
        && request.resource.data.text is string && request.resource.data.text.size() > 0 && request.resource.data.text.size() <= 500;
      allow delete: if isMod();
      allow update: if false;
    }

    match /bannedUsernames/{name} {
      allow read: if signedIn();
      allow create, delete: if isMod();
      allow update: if false;
    }

    match /modLogs/{id} {
      allow read: if isMod();
      allow create: if isMod()
        && request.resource.data.by == request.auth.uid;
      allow update, delete: if false;
    }

    match /anonNotes/{id} {
      allow read: if signedIn() && (resource.data.to == request.auth.uid || isMod());
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['to', 'from', 'text', 'at', 'read'])
        && request.resource.data.from == request.auth.uid
        && request.resource.data.to != request.auth.uid
        && request.resource.data.text is string && request.resource.data.text.size() > 0 && request.resource.data.text.size() <= 280
        && request.resource.data.read == false;
      allow update: if signedIn() && resource.data.to == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['read']);
      allow delete: if isMod() || (signedIn() && resource.data.to == request.auth.uid);
    }

    match /confessions/{id} {
      allow read: if signedIn();
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['text', 'author', 'grade', 'hearts', 'at'])
        && request.resource.data.author == request.auth.uid
        && request.resource.data.text is string
        && request.resource.data.text.size() > 0
        && request.resource.data.text.size() <= 500
        && request.resource.data.hearts == {}
        && request.resource.data.at is int;
      allow update: if signedIn()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['hearts']);
      allow delete: if signedIn() && (resource.data.author == request.auth.uid || isMod());
    }
  }
}
